# Which device each screen accepts (needs states from gun_boot.txt, GUN=1 GUNMODE=lightgun).
# 1. Title "PRESS START": gun trigger / gun Start do nothing, pad 1 Start opens the menu.
load title
gun 160 112 1 5
gun 160 112 8 5
run 60
shot title_after_gun
press S 5
run 60
shot title_after_pad
# 2. Continue: energy forced low, score forced high; at "CONTINUE n" only the dead player's
#    own Start works (gun Start for the Menacer player). Trigger and pad 1 Start are ignored.
load l1
poke FF9EA8 4 00500000
poke FF801E 2 0200
gun 160 100 0 260
save cont
peekeach credits FF80CA 1
peekeach energy FF801E 2
echo cont-trigger
gun 160 100 1 5
gun 160 100 0 60
echo cont-padstart
press S 5
run 60
echo cont-gunstart
gun 160 100 8 5
gun 160 100 0 60
shot cont_after_gunstart
peekeach off
# 3. Let the countdown run out -> GAME OVER -> high-score initials (level $FF80CC = $0A).
